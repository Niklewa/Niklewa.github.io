    // Success callback for popup form (ID must match container id: mlb2-29731922)
    function ml_webform_success_29731922() {
      try { localStorage.setItem('ml_popup_ebook_submitted', '1'); } catch(e){}
      var c = document.getElementById('ebook-popup-content');
      var t = document.getElementById('ebook-popup-thanks');
      if (c) c.classList.add('hidden');
      if (t) t.classList.remove('hidden');
      // Auto-close after a short delay
      setTimeout(function(){ hideEbookPopup(); }, 1800);
    }

    // Popup show/hide + timing
    (function(){
      var overlay = document.getElementById('ebook-popup');
      if (!overlay) return;
      var SHOW_DELAY = 5000; // 5s after page open
      var DISMISS_KEY = 'ml_popup_ebook_hidden_until';
      var SUBMITTED_KEY = 'ml_popup_ebook_submitted';
      var COOL_DOWN_MS = 30*24*60*60*1000; // 30 days suppression
      var now = Date.now();
      var hiddenUntil = 0;
      try { hiddenUntil = parseInt(localStorage.getItem(DISMISS_KEY) || '0', 10); } catch(e) {}
      var submitted = false; try { submitted = !!localStorage.getItem(SUBMITTED_KEY); } catch(e) {}

      // Show only if not suppressed and not submitted
      if (!submitted && now >= hiddenUntil) {
        setTimeout(function(){ showEbookPopup(); }, SHOW_DELAY);
      }

      function showEbookPopup(){
        // Suppression disabled for now
        overlay.classList.remove('hidden');
        overlay.classList.add('flex');
        document.addEventListener('keydown', onKey);
      }

    window.hideEbookPopup = function(setCooldown){
        overlay.classList.add('hidden');
        overlay.classList.remove('flex');
        document.removeEventListener('keydown', onKey);
        if (setCooldown !== false) {
      try { localStorage.setItem(DISMISS_KEY, String(Date.now() + COOL_DOWN_MS)); } catch(e) {}
        }
      }

      function onKey(e){ if (e.key === 'Escape') hideEbookPopup(); }
      // Close interactions
      overlay.querySelector('[data-ml-close="button"]').addEventListener('click', function(){ hideEbookPopup(); });
      overlay.querySelector('[data-ml-close="overlay"]').addEventListener('click', function(){ hideEbookPopup(); });
    })();
  </script>
